import numpy as np
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import PolynomialFeatures
from sklearn.metrics import r2_score

X = np.array([1, 2, 3, 4, 5, 6]).reshape(-1, 1)
y = np.array([2, 5, 10, 17, 26, 37])

linear = LinearRegression()
linear.fit(X, y)
y1 = linear.predict(X)

poly = PolynomialFeatures(degree=2)
X_poly = poly.fit_transform(X)

model = LinearRegression()
model.fit(X_poly, y)
y2 = model.predict(X_poly)

print("Linear Regression R2:", r2_score(y, y1))
print("Polynomial Regression R2:", r2_score(y, y2))

plt.scatter(X, y, color='blue')
plt.plot(X, y1, color='red', label='Linear')
plt.plot(X, y2, color='green', label='Polynomial')
plt.legend()
plt.show()
