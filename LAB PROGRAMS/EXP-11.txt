
import pandas as pd
from sklearn.tree import DecisionTreeClassifier
from sklearn.preprocessing import LabelEncoder

data = {
    'Income': [25000, 50000, 30000, 80000, 60000, 20000],
    'Debt': [20000, 10000, 25000, 5000, 15000, 30000],
    'CreditScore': ['Low', 'High', 'Low', 'High', 'High', 'Low']
}

df = pd.DataFrame(data)

le = LabelEncoder()
y = le.fit_transform(df['CreditScore'])

X = df[['Income', 'Debt']]

model = DecisionTreeClassifier(random_state=42)
model.fit(X, y)

test_data = pd.DataFrame(
    [[70000, 8000]],
    columns=['Income', 'Debt']
)

prediction = model.predict(test_data)

print("Predicted Credit Score:", le.inverse_transform(prediction)[0])
