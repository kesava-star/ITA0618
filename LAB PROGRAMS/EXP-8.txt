
from sklearn.linear_model import LinearRegression

# Training data
X = [[1], [2], [3], [4], [5]]
y = [2, 4, 6, 8, 10]

# Create Linear Regression model
model = LinearRegression()

# Train the model
model.fit(X, y)

# Predict new value
prediction = model.predict([[6]])

print("Predicted Value:", prediction[0])

print("Model Accuracy:", model.score(X, y))
